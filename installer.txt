# OpenStructures desktop release (PowerShell, run from the repository root)
# See docs/FIREBASE_SETUP.md for Firebase setup, migration and release checks.
# The desktop contains public Firebase settings; no MongoDB/SMTP/Admin secrets.
#
# Choose ONE Python version and validate the scientific dependency lock first.
# Existing requirements.txt is not a validated Python 3.13 lock: NumPy 1.26.4
# does not support 3.13, and the newer scientific pins may conflict with it.
# Do not reuse the old Python310 absolute path or delete a working environment.
# Example below uses a NEW disposable build environment; adjust its Python
# version only after resolving and recording a compatible dependency set.

py --list
py -3.12 -m venv .venv-build
$buildPython = Join-Path $PWD '.venv-build\Scripts\python.exe'
& $buildPython --version

# Resolve the repository's dependency pins before continuing if this fails.
& $buildPython -m pip install -r requirements.txt
& $buildPython -m pip install -e . --no-deps
# numba is used by the packaging recipe; record a compatible version in the
# validated build lock if the lock does not already include it.
& $buildPython -m pip install numba
& $buildPython -m pip check

# Once dependencies have been verified, record the exact successful build set:
& $buildPython -m pip freeze > build-requirements.lock.txt

# Copy app/resources/auth_config.example.json to auth_config.json and replace
# placeholders with the project's PUBLIC Firebase web client settings.
# Guest/offline mode remains available if public account config is absent.

& $buildPython -m PyInstaller --noconfirm --clean OpenStructures.spec

# Output: dist\OpenStructures\OpenStructures.exe plus its _internal directory.
